/* --- ENVIAR MENSAGEM E RECEBER RESPOSTA DA IA --- */ async function sendMessage() { const messageText = userInput.value.trim(); if (!messageText) return; // 1. Mostra a mensagem do utilizador no chat const userMsgDiv = document.createElement("div"); userMsgDiv.classList.add("message", "user-message"); userMsgDiv.innerHTML = `
Você
${messageText}`; chatBox.appendChild(userMsgDiv); userInput.value = ""; chatBox.scrollTop = chatBox.scrollHeight; // 2. Cria o balão de "a carregar" da IA const aiMsgDiv = document.createElement("div"); aiMsgDiv.classList.add("message", "assistant-message"); aiMsgDiv.innerHTML = `
CHRONICAL
A pensar...
`; chatBox.appendChild(aiMsgDiv); chatBox.scrollTop = chatBox.scrollHeight; const contentSpan = aiMsgDiv.querySelector(".markdown-content"); try { // 3. Envia o pedido para a API do backend const response = await fetch(BACKEND_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ message: messageText }) }); const data = await response.json(); if (!response.ok) { contentSpan.innerHTML = `${data.error || "Erro no servidor."}`; return; } // 4. Exibe a resposta da IA formatada const responseText = data.text || "Sem resposta."; contentSpan.innerHTML = typeof marked !== "undefined" ? marked.parse(responseText) : responseText; chatBox.scrollTop = chatBox.scrollHeight; } catch (error) { contentSpan.innerHTML = `Erro de conexão ao servidor.`; } }
